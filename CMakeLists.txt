cmake_minimum_required(VERSION 3.16)
project(matmul_free_lm_hls LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MATMULFREE_HLS_SIM "Build native simulation of the HLS decode top" OFF)
option(MATMULFREE_LOCAL_TOOLS "Build private local benchmark/reference tools" OFF)
set(VITIS_ROOT "/opt/xilinx/2025.2/Vitis" CACHE PATH "Vitis installation")

set(HOST_SOURCES src/context.cpp src/decode.cpp src/vocab.cpp src/weight.cpp)

function(configure_host target)
  target_include_directories(${target} PUBLIC src)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-unknown-pragmas
                                           -ffp-contract=off)
  target_link_libraries(${target} PUBLIC m pthread)
endfunction()

add_library(matmulfree_cpu STATIC ${HOST_SOURCES})
target_compile_definitions(matmulfree_cpu PUBLIC USE_CPU_ONLY)
configure_host(matmulfree_cpu)
add_executable(matmulfree src/main.cpp)
target_link_libraries(matmulfree PRIVATE matmulfree_cpu)
if(MATMULFREE_LOCAL_TOOLS)
  foreach(source tests/benchmark.cpp tests/reference_dump.c references/matmul-free-lm.c/run.c)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
      message(FATAL_ERROR "MATMULFREE_LOCAL_TOOLS needs private local scaffolding: ${source}")
    endif()
  endforeach()
  add_executable(matmulfree_bench tests/benchmark.cpp)
  target_link_libraries(matmulfree_bench PRIVATE matmulfree_cpu)
  add_executable(cpu_reference references/matmul-free-lm.c/run.c)
  add_executable(cpu_reference_dump tests/reference_dump.c)
  foreach(target cpu_reference cpu_reference_dump)
    target_compile_options(${target} PRIVATE -ffp-contract=off)
    target_link_libraries(${target} PRIVATE m)
  endforeach()
endif()

if(MATMULFREE_HLS_SIM)
  if(NOT EXISTS "${VITIS_ROOT}/include/hls_stream.h")
    message(FATAL_ERROR "Set VITIS_ROOT to a Vitis installation for HLS simulation")
  endif()
  add_library(decode_kernel OBJECT src/decode.cpp)
  target_compile_definitions(decode_kernel PRIVATE BUILD_DECODE_KERNEL USE_NATIVE_MATH)
  target_include_directories(decode_kernel PRIVATE src "${VITIS_ROOT}/include")
  target_compile_options(decode_kernel PRIVATE -ffp-contract=off -Wno-unknown-pragmas)
  add_library(matmulfree_sim_runtime STATIC ${HOST_SOURCES})
  target_compile_definitions(matmulfree_sim_runtime PUBLIC USE_CPU_ONLY USE_HLS_SIM)
  configure_host(matmulfree_sim_runtime)
  add_executable(matmulfree_sim src/main.cpp $<TARGET_OBJECTS:decode_kernel>)
  target_link_libraries(matmulfree_sim PRIVATE matmulfree_sim_runtime)
  if(MATMULFREE_LOCAL_TOOLS)
    add_executable(matmulfree_sim_bench tests/benchmark.cpp $<TARGET_OBJECTS:decode_kernel>)
    target_link_libraries(matmulfree_sim_bench PRIVATE matmulfree_sim_runtime)
  endif()
endif()
